function doGet(e) { var output = buildResponse(e); return output; } function doPost(e) { var output = handlePost(e); return output; } function buildResponse(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var ventas = [], reclutas = [], comV = 1800, meta = 100000; var vs = ss.getSheetByName('Ventas'); if (vs && vs.getLastRow() > 1) { var vd = vs.getRange(2,1,vs.getLastRow()-1,3).getValues(); ventas = vd.map(function(r){ return {fecha:r[0].toString(), monto:Number(r[1]), nota:r[2]}; }); } var rs = ss.getSheetByName('Reclutas'); if (rs && rs.getLastRow() > 1) { var rd = rs.getRange(2,1,rs.getLastRow()-1,2).getValues(); reclutas = rd.map(function(r){ return {fecha:r[0].toString(), nombre:r[1].toString()}; }); } var cs = ss.getSheetByName('Config'); if (cs && cs.getLastRow() > 1) { var cd = cs.getRange(2,1,1,2).getValues()[0]; comV = Number(cd[0]) || 1800; meta = Number(cd[1]) || 100000; } var json = JSON.stringify({ ok:true, ventas:ventas, reclutas:reclutas, comV:comV, meta:meta }); return ContentService.createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput( JSON.stringify({ok:false, error:err.message}) ).setMimeType(ContentService.MimeType.JSON); } } function handlePost(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var data = JSON.parse(e.postData.contents); var action = data.action; if (action === 'addVenta') { var sh = ss.getSheetByName('Ventas') || ss.insertSheet('Ventas'); if (sh.getLastRow() === 0) sh.appendRow(['Fecha','Monto','Nota']); sh.appendRow([data.fecha, data.monto, data.nota||'']); } if (action === 'addRecluta') { var sh = ss.getSheetByName('Reclutas') || ss.insertSheet('Reclutas'); if (sh.getLastRow() === 0) sh.appendRow(['Fecha','Nombre']); sh.appendRow([data.fecha, data.nombre]); } if (action === 'delRecluta') { var sh = ss.getSheetByName('Reclutas'); if (sh) { var rows = sh.getDataRange().getValues(); for (var i = rows.length-1; i >= 1; i--) { if (rows[i][1] === data.nombre) { sh.deleteRow(i+1); break; } } } } if (action === 'saveConfig') { var sh = ss.getSheetByName('Config') || ss.insertSheet('Config'); sh.clearContents(); sh.appendRow(['comV','meta']); sh.appendRow([data.comV, data.meta]); } if (action === 'reset') { ['Ventas','Reclutas'].forEach(function(n){ var s = ss.getSheetByName(n); if(s) s.clearContents(); }); } return ContentService.createTextOutput( JSON.stringify({ok:true}) ).setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput( JSON.stringify({ok:false, error:err.message}) ).setMimeType(ContentService.MimeType.JSON); } }
